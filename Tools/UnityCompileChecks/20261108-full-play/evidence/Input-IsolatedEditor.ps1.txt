param([ValidateSet('Capture','Click','Key')][string]$Action='Capture',[int]$X=0,[int]$Y=0,[string]$Keys='')
$ErrorActionPreference='Stop'
$editorProcess=Get-Process -Id 708
$processInfo=Get-CimInstance Win32_Process -Filter 'ProcessId=708'
if($processInfo.CommandLine -notlike '*26ef9011-validation*source.local*'){throw 'Wrong Editor project'}
Add-Type -AssemblyName System.Drawing
Add-Type -AssemblyName System.Windows.Forms
Add-Type @'
using System; using System.Runtime.InteropServices;
public static class PlayEditorWindow {
 [StructLayout(LayoutKind.Sequential)] public struct RECT {public int Left,Top,Right,Bottom;}
 [DllImport("user32.dll")] public static extern bool GetWindowRect(IntPtr h,out RECT r);
 [DllImport("user32.dll")] public static extern bool ShowWindow(IntPtr h,int cmd);
 [DllImport("user32.dll")] public static extern bool SetForegroundWindow(IntPtr h);
 [DllImport("user32.dll")] public static extern IntPtr GetForegroundWindow();
 [DllImport("user32.dll")] public static extern uint GetWindowThreadProcessId(IntPtr h,out uint pid);
 [DllImport("user32.dll")] public static extern bool SetCursorPos(int x,int y);
 [DllImport("user32.dll")] public static extern void mouse_event(uint flags,uint dx,uint dy,uint data,UIntPtr extra);
}
'@
$handle=$editorProcess.MainWindowHandle
if($handle -eq [IntPtr]::Zero){throw 'Editor has no window'}
[void][PlayEditorWindow]::ShowWindow($handle,3)
[void][PlayEditorWindow]::SetForegroundWindow($handle)
Start-Sleep -Milliseconds 350
[uint32]$foregroundProcess=0
[void][PlayEditorWindow]::GetWindowThreadProcessId([PlayEditorWindow]::GetForegroundWindow(),[ref]$foregroundProcess)
if($foregroundProcess -ne 708){throw 'Could not focus isolated Editor; refusing input'}
$rect=New-Object PlayEditorWindow+RECT
[void][PlayEditorWindow]::GetWindowRect($handle,[ref]$rect)
$width=$rect.Right-$rect.Left; $height=$rect.Bottom-$rect.Top
if($Action -eq 'Click'){
 if($X -lt 0 -or $Y -lt 0 -or $X -ge $width -or $Y -ge $height){throw 'Click outside verified window'}
 [void][PlayEditorWindow]::SetCursorPos($rect.Left+$X,$rect.Top+$Y)
 [PlayEditorWindow]::mouse_event(2,0,0,0,[UIntPtr]::Zero)
 Start-Sleep -Milliseconds 100
 [PlayEditorWindow]::mouse_event(4,0,0,0,[UIntPtr]::Zero)
} elseif($Action -eq 'Key') {
 if([string]::IsNullOrEmpty($Keys)){throw 'Empty key sequence'}
 [System.Windows.Forms.SendKeys]::SendWait($Keys)
} else {
 $bitmap=New-Object System.Drawing.Bitmap($width,$height)
 $graphics=[System.Drawing.Graphics]::FromImage($bitmap)
 try {$graphics.CopyFromScreen($rect.Left,$rect.Top,0,0,$bitmap.Size); $bitmap.Save((Join-Path $PSScriptRoot 'editor-window.local.png'),[System.Drawing.Imaging.ImageFormat]::Png)}
 finally {$graphics.Dispose();$bitmap.Dispose()}
}
[pscustomobject]@{Action=$Action;Pid=708;Width=$width;Height=$height;X=$X;Y=$Y;Keys=$Keys;ProjectVerified=$true;Focused=$true}|ConvertTo-Json -Compress
